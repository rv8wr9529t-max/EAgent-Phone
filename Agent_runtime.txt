function Add-Event([string]$who, [string]$text) {
    $ev = @{ t = (Get-Date -Format 'HH:mm:ss'); who = $who; text = $text }
    [System.Threading.Monitor]::Enter($State.SyncRoot)
    try {
        $State.Events.Add($ev) | Out-Null
        while ($State.Events.Count -gt 400) { $State.Events.RemoveAt(0) }
    } finally { [System.Threading.Monitor]::Exit($State.SyncRoot) }
    $color = switch ($who) { 'user' { 'Cyan' } 'ai' { 'Green' } 'tool' { 'Yellow' } default { 'DarkGray' } }
    $short = if ($text.Length -gt 120) { $text.Substring(0, 120) + '…' } else { $text }
    Write-Host ("[{0}] {1}: {2}" -f $ev.t, $who.ToUpper(), ($short -replace "`r?`n", ' | ')) -ForegroundColor $color
}

function New-CdpSession {
    $pages = Invoke-RestMethod -Uri "http://127.0.0.1:$CdpPort/json" -TimeoutSec 5
    $target = $pages | Where-Object { $_.url -match 'chat\.deepseek\.com' -and $_.type -eq 'page' } | Select-Object -First 1
    if (-not $target) {
        try {
            $null = Invoke-RestMethod -Uri "http://127.0.0.1:$CdpPort/json/new?https://chat.deepseek.com" -Method Put -TimeoutSec 6
            Start-Sleep -Seconds 2
            $pages = Invoke-RestMethod -Uri "http://127.0.0.1:$CdpPort/json" -TimeoutSec 5
            $target = $pages | Where-Object { $_.url -match 'chat\.deepseek\.com' -and $_.type -eq 'page' } | Select-Object -First 1
        } catch {}
    }
    if (-not $target) { return $null }
    $ws = New-Object System.Net.WebSockets.ClientWebSocket
    $ws.ConnectAsync([Uri]$target.webSocketDebuggerUrl, [Threading.CancellationToken]::None).Wait(5000) | Out-Null
    if ($ws.State -ne 'Open') { $ws.Dispose(); return $null }
    return @{ ws = $ws; id = 0 }
}

function Cdp-Recv($s, [int]$id, [int]$timeoutMs) {
    $buf = New-Object byte[] 1048576
    $recvSeg = New-Object System.ArraySegment[byte] -ArgumentList @(,$buf)
    $deadline = (Get-Date).AddMilliseconds($timeoutMs)
    while ((Get-Date) -lt $deadline) {
        $msLeft = [int]($deadline - (Get-Date)).TotalMilliseconds
        if ($msLeft -lt 1) { return $null }
        $task = $s.ws.ReceiveAsync($recvSeg, [Threading.CancellationToken]::None)
        if (-not $task.Wait($msLeft)) { return $null }
        $rr = $task.Result
        if ($rr -and $rr.Count -gt 0) {
            $text = [Text.Encoding]::UTF8.GetString($buf, 0, $rr.Count)
            $pattern = '"id"\s*:\s*' + $id + '\b'
            foreach ($line in ($text -split "`n")) {
                if ($line -match $pattern) { try { return ($line | ConvertFrom-Json) } catch {} }
            }
        }
    }
    return $null
}

function Cdp-Call($s, [string]$method, $params, [int]$timeoutMs = 8000) {
    $s.id++; $id = $s.id
    $payload = @{ id = $id; method = $method; params = $params } | ConvertTo-Json -Depth 20 -Compress
    $bytes = [Text.Encoding]::UTF8.GetBytes($payload)
    $seg = New-Object System.ArraySegment[byte] -ArgumentList @(,$bytes)
    $s.ws.SendAsync($seg, [System.Net.WebSockets.WebSocketMessageType]::Text, $true, [Threading.CancellationToken]::None).Wait(3000) | Out-Null
    $resp = Cdp-Recv $s $id $timeoutMs
    if ($resp -and $resp.result) { return $resp.result }
    return $null
}

function Cdp-Eval($s, [string]$expr, [int]$timeoutMs = 8000) {
    $r = Cdp-Call $s 'Runtime.evaluate' @{ expression = $expr; returnByValue = $true } $timeoutMs
    if ($r -and $r.result) { return $r.result.value }
    return $null
}

function Send-Chat([string]$Message, [string[]]$Files = @()) {

    try {

        $null = Cdp-Call $s 'Page.bringToFront' @{}
        $null = Cdp-Call $s 'Emulation.setFocusEmulationEnabled' @{ enabled = $true }

        $msgJs = $SelJs + @'
(function(){
    var els = pick('assistant');
    if(els.length===0) return JSON.stringify({count:0, text:''});
    var last = els[els.length-1];
    return JSON.stringify({count:els.length, text:(last.textContent||'').trim()});
})()
'@
        $lastAssist0 = Cdp-Eval $s $msgJs 4000
        $lastAssistJs = $SelJs + @'
(function(){
    var a=pick('assistant');
    if(a.length===0) return '';
    return (a[a.length-1].textContent||'').trim();
})()
'@
        $lastAssist0 = Cdp-Eval $s $lastAssistJs 4000
        if ($null -eq $lastAssist0) { $lastAssist0 = '' }

        if ($Files.Count -gt 0) {
            $abs = @()
            foreach ($f in $Files) { if (Test-Path -LiteralPath $f) { $abs += (Resolve-Path -LiteralPath $f).Path } }
            if ($abs.Count -gt 0) {
                $null = Cdp-Call $s 'DOM.enable' @{}
                $objRes = Cdp-Call $s 'Runtime.evaluate' @{ expression = 'document.querySelector(''input[type="file"]'')'; returnByValue = $false }
                if ($objRes -and $objRes.result -and $objRes.result.objectId) {
                    $null = Cdp-Call $s 'DOM.setFileInputFiles' @{ files = $abs; objectId = $objRes.result.objectId }
                    Start-Sleep -Seconds 3
                } else { Add-Event 'sys' '页面找不到文件输入框，附件未上传' }
            }
        }

        if ($Message) {
            $msgJson = $Message | ConvertTo-Json -Compress
            $injectJs = $SelJs + @"
(function(){
    var msg = $msgJson;
    var ta = pick('input')[0];
    if(!ta) return JSON.stringify({ok:false, err:'no textarea'});
    ta.focus();
    var s = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, 'value').set;
    s.call(ta, msg);
    ta.dispatchEvent(new Event('input',{bubbles:true}));
    ta.dispatchEvent(new Event('change',{bubbles:true}));
    return JSON.stringify({ok:true, len:ta.value.length});
})()
"@
            $r = Cdp-Eval $s $injectJs
            if (-not $r -or $r -notmatch '"ok":true') { Add-Event 'sys' "消息注入失败: $r"; return $null }
            Start-Sleep -Milliseconds 300
            $sendJs = $SelJs + @'
(function(){
    var ta = pick('input')[0];
    if(!ta) return 'NO_TA';
    ta.focus();
    function ev(t){
        var e = new KeyboardEvent(t, {key:'Enter', code:'Enter', keyCode:13, which:13, bubbles:true, cancelable:true});
        ta.dispatchEvent(e);
    }
    ev('keydown'); ev('keypress'); ev('keyup');
    return 'SENT';
})()
'@
            $r = Cdp-Eval $s $sendJs
            if ($r -ne 'SENT') { Add-Event 'sys' "发送动作失败: $r"; return $null }
        }

        $deadline = (Get-Date).AddSeconds($ReplyTimeoutSec)
        $finalText = ''
        $prevText = $lastAssist0
        $stableTicks = 0
        $requiredStable = 7
        $sawNew = $false
        while ((Get-Date) -lt $deadline) {
            Start-Sleep -Milliseconds 300
            if ($State.Stop) { return $null }
            $cur = Cdp-Eval $s $lastAssistJs 5000
            if ($null -eq $cur) { continue }
            $cur = [string]$cur
            if (-not $sawNew) {
                if ($cur -ne $lastAssist0 -and $cur.Length -gt 0) { $sawNew = $true; $prevText = $cur }
                continue
            }
            if ($cur -eq $prevText) {
                $stableTicks++
                if ($stableTicks -ge $requiredStable) { $finalText = $cur; break }
            } else { $stableTicks = 0; $prevText = $cur }
        }
        if (-not $finalText) { if (-not $State.Stop) { Add-Event 'sys' "等待 AI 回复超时($ReplyTimeoutSec 秒)" }; return $null }
        return $finalText
    } catch {
        Add-Event 'sys' ("CDP 通信异常: " + $_.Exception.Message)
        return $null
    } finally {
        try { $ws.Dispose() } catch {}
    }
}

# ---------------- 工具标记解析 ----------------
function Parse-ToolCall([string]$reply) {
    if ($reply -match '(?s)# powershell_tool_调用工具_识别头.*?# powershell_tool_调用工具_识别尾[^\r\n]*') {
        return @{ Type = 'powershell_tool'; Code = $matches[0] }
    }
    if ($reply -match 'agent_getjpg_1_star\((?<a>[^)]*)\)_1_getjpg_end') {
        return @{ Type = 'agent_getjpg'; Target = $matches['a'].Trim() }
    }
    if ($reply -match '(?s)agent_getfile_head\((?<a>.*?)\)agent_getfile_end') {
        $paths = @()
        foreach ($qm in [regex]::Matches($matches['a'], "'([^']+)'")) { $paths += $qm.Groups[1].Value }
        return @{ Type = 'agent_getfile'; Paths = $paths }
    }
    if ($reply -match 'agent_mouse_simulate_1_start\((\d+)\)_agent_mouse_simulate_1_end') {
        $x = [int]$matches[1]; $y = 0; $btn = 0
        if ($reply -match 'agent_mouse_simulate_2_start\((\d+)\)_agent_mouse_simulate_2_end') { $y = [int]$matches[1] }
        if ($reply -match 'agent_mouse_simulate_3_start\((\d+)\)_agent_mouse_simulate_3_end') { $btn = [int]$matches[1] }
        return @{ Type = 'agent_mouse_simulate'; X = $x; Y = $y; Btn = $btn }
    }
    if ($reply -match 'agent_keyboard_simulate') {
        $vk1 = 0; $vk2 = 0; $vk3 = 0; $hwnd = 0
        if ($reply -match 'agent_keyboard_simulate_1_start\((\d+)\)_agent_keyboard_simulate_1_end') { $vk1 = [int]$matches[1] }
        if ($reply -match 'agent_keyboard_simulate_2_start\((\d+)\)_agent_keyboard_simulate_2_end') { $vk2 = [int]$matches[1] }
        if ($reply -match 'agent_keyboard_simulate_3_start\((\d+)\)_agent_keyboard_simulate_3_end') { $vk3 = [int]$matches[1] }
        if ($reply -match 'agent_keyboard_simulate_4_start\((\d+)\)_agent_keyboard_simulate_4_end') { $hwnd = [int]$matches[1] }
        return @{ Type = 'agent_keyboard_simulate'; VK1 = $vk1; VK2 = $vk2; VK3 = $vk3; Hwnd = $hwnd }
    }
    return $null
}

# ---------------- 手机确认 ----------------
function Wait-Confirm([string]$tool, [string]$preview) {
    $State.ConfirmResult = $null
    $State.PendingConfirm = @{ id = [guid]::NewGuid().ToString('N'); tool = $tool; preview = $preview }
    $State.Status = '等待手机确认'
    Add-Event 'tool' ("等待确认: " + $tool)
    $deadline = (Get-Date).AddSeconds(120)
    $result = $false
    while ((Get-Date) -lt $deadline) {
        if ($null -ne $State.ConfirmResult) { $result = [bool]$State.ConfirmResult; break }
        if ($State.Stop) { break }
        Start-Sleep -Milliseconds 300
    }
    $State.PendingConfirm = $null
    Add-Event 'tool' ($(if ($result) { '✅ 已允许' } else { '🚫 已拒绝/超时' }))
    return $result
}

# ---------------- 工具执行器 ----------------
function Invoke-PSTool([string]$code) {
    if ($code.Length -gt 3800) {
        return @{ Feedback = "!![工具][错误][PowerShell]!! 代码超过3800字限制`n//[END]//"; Files = @(); Abort = $false }
    }
    $preview = ($code -split "`n" | Select-Object -First 14) -join "`n"
    if ($code.Length -gt 800) { $preview += "`n…(共 $($code.Length) 字)" }
    if (-not (Wait-Confirm '执行PowerShell' $preview)) {
        return @{ Feedback = "[工具]用户拒绝执行命令`n//[END]//"; Files = @(); Abort = $false }
    }
    $psFile = Join-Path $TempDir ("ps_{0}.ps1" -f ([guid]::NewGuid().ToString('N').Substring(0, 8)))
    $outFile = Join-Path $TempDir ("ps_{0}_out.txt" -f ([guid]::NewGuid().ToString('N').Substring(0, 8)))
    [IO.File]::WriteAllText($psFile, $code, (New-Object System.Text.UTF8Encoding($true)))
    Add-Event 'tool' "执行 PowerShell ($($code.Length) 字)"
    $argStr = "-NoProfile -ExecutionPolicy Bypass -Command `"& '$psFile' *>&1 | Out-File -FilePath '$outFile' -Encoding utf8`""
    $proc = Start-Process -FilePath 'powershell.exe' -ArgumentList $argStr -WindowStyle Hidden -PassThru
    $exited = $proc.WaitForExit(120000)
    if (-not $exited) {
        try { & taskkill /PID $proc.Id /T /F 2>$null | Out-Null } catch {}
        $out = '[超时120秒，进程已强制终止]'
    } else {
        Start-Sleep -Milliseconds 300
        $out = ''
        if (Test-Path $outFile) { try { $out = [IO.File]::ReadAllText($outFile, [Text.Encoding]::UTF8) } catch { $out = "(读取输出失败)" } }
        $out = $out.TrimEnd()
        if (-not $out) { $out = '（无输出）' }
    }
    if ($out.Length -gt 3400) {
        $headLen = 1700; $tailLen = 1500
        $out = $out.Substring(0, $headLen) + "`n…[输出过长已截断,共 $($out.Length) 字]…`n" + $out.Substring($out.Length - $tailLen)
    }
    Remove-Item $psFile, $outFile -Force -ErrorAction SilentlyContinue
    Add-Event 'tool' ("PowerShell 结果: " + $(if ($out.Length -gt 200) { $out.Substring(0, 200) + '…' } else { $out }))
    return @{ Feedback = "//[工具][取回结果]//`n$out`n//[END]//"; Files = @(); Abort = $false }
}

function Save-ScreenShot([string]$target) {
    $jpgPath = Join-Path $TempDir '屏幕截图.jpg'
    $bmp = $null
    if ($target -match '^\d+$' -and [int64]$target -gt 0) {
        $h = [IntPtr]([int64]$target)
        $rc = New-Object RECT
        if ([U32]::GetWindowRect($h, [ref]$rc)) {
            $w = $rc.Right - $rc.Left; $hh = $rc.Bottom - $rc.Top
            if ($w -gt 10 -and $hh -gt 10) {
                try {
                    $t = New-Object System.Drawing.Bitmap($w, $hh)
                    $g = [System.Drawing.Graphics]::FromImage($t)
                    $hdc = $g.GetHdc()
                    $ok = [U32]::PrintWindow($h, $hdc, 2)
                    $g.ReleaseHdc(); $g.Dispose()
                    if ($ok) { $bmp = $t } else { $t.Dispose() }
                } catch {}
            }
        }
    }
    if (-not $bmp) {
        $x = [U32]::GetSystemMetrics(76); $y = [U32]::GetSystemMetrics(77)
        $w = [U32]::GetSystemMetrics(78); $h = [U32]::GetSystemMetrics(79)
        $bmp = New-Object System.Drawing.Bitmap($w, $h)
        $g = [System.Drawing.Graphics]::FromImage($bmp)
        $g.CopyFromScreen($x, $y, 0, 0, $bmp.Size)
        $g.Dispose()
    }
    $codec = [System.Drawing.Imaging.ImageCodecInfo]::GetImageEncoders() | Where-Object { $_.MimeType -eq 'image/jpeg' }
    $ep = New-Object System.Drawing.Imaging.EncoderParameters(1)
    $ep.Param[0] = New-Object System.Drawing.Imaging.EncoderParameter([System.Drawing.Imaging.Encoder]::Quality, [long]80)
    if (Test-Path $jpgPath) { Remove-Item $jpgPath -Force }
    $bmp.Save($jpgPath, $codec, $ep)
    $bmp.Dispose()
    return $jpgPath
}

function Invoke-GetJpg([string]$target) {
    Add-Event 'tool' "截图: $target"
    $jpg = Save-ScreenShot $target
    $pt = New-Object POINT
    [U32]::GetCursorPos([ref]$pt) | Out-Null
    $lw = [U32]::GetSystemMetrics(0); $lh = [U32]::GetSystemMetrics(1)
    $rr = 60
    try { $vc = Get-CimInstance Win32_VideoController -ErrorAction Stop | Select-Object -First 1; if ($vc.CurrentRefreshRate) { $rr = $vc.CurrentRefreshRate } } catch {}
    $fb = "//[工具][屏幕截图]//【提示：如果涉及鼠标操控，优先用逻辑坐标，而不是物理坐标】，当前鼠标所在的屏幕逻辑坐标XY为：($($pt.X),$($pt.Y))，屏幕刷新率=$rr，物理屏幕宽度=$lw ->逻辑宽度=$lw，物理屏幕高度=$lh ->逻辑高度=$lh`n//[END]//"
    return @{ Feedback = $fb; Files = @($jpg); Abort = $false }
}

function Invoke-GetFile([string[]]$paths) {
    if ($paths.Count -eq 0) { return @{ Feedback = "!![工具][错误][上传文件]!! 未解析到文件路径`n//[END]//"; Files = @(); Abort = $false } }
    if ($paths.Count -gt 50) { return @{ Feedback = "!![工具][错误][上传文件]!! 一次最多50个文件`n//[END]//"; Files = @(); Abort = $false } }
    $ok = @(); $bad = @()
    foreach ($p in $paths) {
        if (-not (Test-Path -LiteralPath $p -PathType Leaf)) { $bad += "$p (不存在)"; continue }
        $len = (Get-Item -LiteralPath $p).Length
        if ($len -gt 100MB) { $bad += "$p (超过100MB)"; continue }
        $ok += (Resolve-Path -LiteralPath $p).Path
    }
    Add-Event 'tool' ("上传文件: " + ($ok -join '; '))
    $fb = "//[工具][上传文件]//已请求上传 $($ok.Count) 个文件"
    if ($bad.Count -gt 0) { $fb += "；以下文件失败: " + ($bad -join '; ') }
    $fb += "`n//[END]//"
    return @{ Feedback = $fb; Files = $ok; Abort = $false }
}

function Invoke-Mouse([int]$x, [int]$y, [int]$btn) {
    Add-Event 'tool' "鼠标: ($x,$y) btn=$btn"
    [U32]::SetCursorPos($x, $y) | Out-Null
    Start-Sleep -Milliseconds 120
    $LDOWN = 0x02; $LUP = 0x04; $RDOWN = 0x08; $RUP = 0x10
    $click = {
        param($down, $up, $times)
        for ($i = 0; $i -lt $times; $i++) {
            [U32]::mouse_event($down, 0, 0, 0, [IntPtr]::Zero)
            Start-Sleep -Milliseconds 60
            [U32]::mouse_event($up, 0, 0, 0, [IntPtr]::Zero)
            Start-Sleep -Milliseconds 110
        }
    }
    switch ($btn) {
        0 { & $click $LDOWN $LUP 1 }
        1 { & $click $LDOWN $LUP 2 }
        2 { & $click $RDOWN $RUP 1 }
        3 { & $click $RDOWN $RUP 2 }
    }
    $fb = "//[工具][模拟控制鼠标]//已执行(除非用户移动鼠标)。当前鼠标所在的屏幕逻辑坐标XY为：($x,$y)`n//[END]//"
    return @{ Feedback = $fb; Files = @(); Abort = $false }
}

function Invoke-KeyboardTool([int]$vk1, [int]$vk2, [int]$vk3, [int]$hwnd) {
    Add-Event 'tool' "键盘: $vk1,$vk2,$vk3 hwnd=$hwnd"
    if ($hwnd -gt 0) {
        $h = [IntPtr]$hwnd
        [U32]::ShowWindow($h, 9) | Out-Null
        [U32]::keybd_event(0x12, 0, 0, [UIntPtr]::Zero)
        Start-Sleep -Milliseconds 40
        [U32]::keybd_event(0x12, 0, 2, [UIntPtr]::Zero)
        Start-Sleep -Milliseconds 40
        [U32]::SetForegroundWindow($h) | Out-Null
        Start-Sleep -Milliseconds 250
    }
    $KEYUP = 2
    if ($vk1 -gt 0) { [U32]::keybd_event([byte]$vk1, 0, 0, [UIntPtr]::Zero); Start-Sleep -Milliseconds 45 }
    if ($vk2 -gt 0) { [U32]::keybd_event([byte]$vk2, 0, 0, [UIntPtr]::Zero); Start-Sleep -Milliseconds 45 }
    if ($vk3 -gt 0) { [U32]::keybd_event([byte]$vk3, 0, 0, [UIntPtr]::Zero); Start-Sleep -Milliseconds 45 }
    if ($vk3 -gt 0) { [U32]::keybd_event([byte]$vk3, 0, $KEYUP, [UIntPtr]::Zero); Start-Sleep -Milliseconds 35 }
    if ($vk2 -gt 0) { [U32]::keybd_event([byte]$vk2, 0, $KEYUP, [UIntPtr]::Zero); Start-Sleep -Milliseconds 35 }
    if ($vk1 -gt 0) { [U32]::keybd_event([byte]$vk1, 0, $KEYUP, [UIntPtr]::Zero) }
    $fb = "//[工具][模拟控制键盘]//已执行。`n//[END]//"
    return @{ Feedback = $fb; Files = @(); Abort = $false }
}

function Execute-Tool($tc) {
    switch ($tc.Type) {
        'powershell_tool' { return (Invoke-PSTool $tc.Code) }
        'agent_getjpg' {
            if ($ConfirmAll -and -not (Wait-Confirm '屏幕截图' "目标: $($tc.Target)")) { return @{ Feedback = "[工具]用户拒绝执行命令`n//[END]//"; Files = @(); Abort = $false } }
            return (Invoke-GetJpg $tc.Target)
        }
        'agent_getfile' {
            if ($ConfirmAll -and -not (Wait-Confirm '上传文件' ($tc.Paths -join "`n"))) { return @{ Feedback = "[工具]用户拒绝执行命令`n//[END]//"; Files = @(); Abort = $false } }
            return (Invoke-GetFile $tc.Paths)
        }
        'agent_mouse_simulate' {
            if ($ConfirmAll -and -not (Wait-Confirm '模拟鼠标' "X=$($tc.X) Y=$($tc.Y) Btn=$($tc.Btn)")) { return @{ Feedback = "[工具]用户拒绝执行命令`n//[END]//"; Files = @(); Abort = $false } }
            return (Invoke-Mouse $tc.X $tc.Y $tc.Btn)
        }
        'agent_keyboard_simulate' {
            if ($ConfirmAll -and -not (Wait-Confirm '模拟键盘' "VK=$($tc.VK1),$($tc.VK2),$($tc.VK3) Hwnd=$($tc.Hwnd)")) { return @{ Feedback = "[工具]用户拒绝执行命令`n//[END]//"; Files = @(); Abort = $false } }
            return (Invoke-KeyboardTool $tc.VK1 $tc.VK2 $tc.VK3 $tc.Hwnd)
        }
    }
    return @{ Feedback = "!![工具][错误]!! 未知工具`n//[END]//"; Files = @(); Abort = $false }
}

function Process-Command([string]$text) {
    $State.Busy = $true
    Add-Event 'user' $text
    try {
        if ($script:NeedSystemPrompt) {
            $State.Status = '初始化会话(发系统提示词)'
            $r = Send-Chat $sysPrompt
            if ($null -eq $r) { Add-Event 'sys' '系统提示词发送失败'; $State.Status = '失败'; return }
            Add-Event 'ai' $r
            $script:NeedSystemPrompt = $false
        }
        $msg = $text
        [string[]]$files = @()
        for ($round = 1; $round -le $MaxToolRounds; $round++) {
            if ($State.Stop) { return }
            $State.Status = "等待AI回复 (第$round轮)"
            $reply = Send-Chat $msg $files
            if ($null -eq $reply) { $State.Status = '失败'; return }
            Add-Event 'ai' $reply
            $tc = Parse-ToolCall $reply
            if (-not $tc) { $State.Status = '完成'; break }
            $State.Status = "执行工具: $($tc.Type)"
            $res = Execute-Tool $tc
            if ($res.Abort) { $State.Status = '已中止'; break }
            $msg = $res.Feedback
            [string[]]$files = @($res.Files)
            if ($round -eq $MaxToolRounds) { Add-Event 'sys' "达到最大工具轮数 $MaxToolRounds，停止续接"; $State.Status = '达到轮数上限' }
        }
    } finally {
        $State.Busy = $false
        if ($State.Status -match '第\d+轮|执行工具|初始化') { $State.Status = '空闲' }
    }
}

while (-not $State.Stop) {
    if ($State.Reset) {
        $State.Reset = $false
        $script:NeedSystemPrompt = $true
        [System.Threading.Monitor]::Enter($State.SyncRoot)
        try { $State.Queue.Clear() } finally { [System.Threading.Monitor]::Exit($State.SyncRoot) }
        try {
            $s = New-CdpSession
            if ($s) {
                Cdp-Eval $s "location.href='https://chat.deepseek.com/'" 4000 | Out-Null
                $s.ws.Dispose()
            }
        } catch {}
        Add-Event 'sys' '已切换新会话，下一条命令会重建上下文'
        $State.Status = '空闲'
    }
    if ($State.Queue.Count -gt 0 -and -not $State.Busy) {
        $cmd = $null
        [System.Threading.Monitor]::Enter($State.SyncRoot)
        try { if ($State.Queue.Count -gt 0) { $cmd = $State.Queue[0]; $State.Queue.RemoveAt(0) } }
        finally { [System.Threading.Monitor]::Exit($State.SyncRoot) }
        if ($cmd) {
            if (-not (Ensure-Browser)) { Add-Event 'sys' '浏览器不可用，命令已丢弃' }
            else { Process-Command $cmd }
        }
        continue
    }
    try {
        if ([Console]::KeyAvailable) {
            $line = Read-Host '本机命令'
            if ($line.Trim()) {
                [System.Threading.Monitor]::Enter($State.SyncRoot)
                try { $State.Queue.Add($line.Trim()) | Out-Null } finally { [System.Threading.Monitor]::Exit($State.SyncRoot) }
            }
        }
    } catch {}
    Start-Sleep -Milliseconds 250
}
